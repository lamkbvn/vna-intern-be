ROLLBACK;

BEGIN;

-- Tạo enum mới
CREATE TYPE report_period_enum_new AS ENUM (
    '6 months start year',
    '6 months end year',
    '12 months'
);

-- Chuyển enum cho bảng report
ALTER TABLE report
ALTER COLUMN period TYPE report_period_enum_new
USING period::text::report_period_enum_new;

-- Chuyển enum cho bảng report_config
ALTER TABLE report_config
ALTER COLUMN period TYPE report_period_enum_new
USING period::text::report_period_enum_new;

-- Xóa enum cũ
DROP TYPE report_period_enum;

-- Đổi tên enum mới
ALTER TYPE report_period_enum_new
RENAME TO report_period_enum;

COMMIT;
